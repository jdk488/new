    "import time\n",
    "\n",
    "def chatbot():\n",
    "    knowledge_base = {\n",
    "        \"hello\": \"Hello! Welcome to PDEA Cafe. How can I help you today?\",\n",
    "        \"hi\": \"Hi there! Looking for a table or checking our menu?\",\n",
    "        \"menu\": \"Today we have:\\n1. Coffee\\n2. Sandwich\\n3. Pasta\\nWould you like to order something?\",\n",
    "        \"order\": \"Sure! Please tell me your items, and I will note them down.\",\n",
    "        \"hours\": \"We are open from 9:00 AM to 10:00 PM, Monday to Saturday.\",\n",
    "        \"location\": \"We are located at Manjari (Bk.), Hadapsar, Pune.\",\n",
    "        \"bye\": \"Thank you for visiting PDEA Cafe! Have a great day!\",\n",
    "        \"thanks\": \"You're very welcome!\",\n",
    "        \"help\": \"I can help you with the menu, our timings, location, or taking an order.\"\n",
    "    }\n",
    "\n",
    "    print(\"--- PDEA Elementary Chatbot ---\")\n",
    "    print(\"Type 'bye' to exit the conversation.\\n\")\n",
    "\n",
    "    while True:\n",
    "        user_input = input(\"You: \").lower().strip()\n",
    "\n",
    "        if user_input == 'bye':\n",
    "            print(f\"Chatbot: {knowledge_base['bye']}\")\n",
    "            break\n",
    "\n",
    "        found_response = False\n",
    "\n",
    "        for key in knowledge_base:\n",
    "            if key in user_input:\n",
    "                print(f\"Chatbot: {knowledge_base[key]}\")\n",
    "                found_response = True\n",
    "                break\n",
    "\n",
    "        if not found_response:\n",
    "            print(\"Chatbot: I'm sorry, I didn't quite get that. Could you rephrase? (Type 'help' for options)\")\n",
    "\n",
    "\n",
    "if __name__ == \"__main__\":\n",
    "    chatbot()"
