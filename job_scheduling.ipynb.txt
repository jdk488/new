    "def job_scheduling(jobs):\n",
    "    jobs.sort(key=lambda x: x[2], reverse=True)\n",
    "    max_deadline = max(job[1] for job in jobs)\n",
    "\n",
    "    slots = [-1] * (max_deadline + 1)\n",
    "    profit = 0\n",
    "\n",
    "    for job in jobs:\n",
    "        for j in range(job[1], 0, -1):\n",
    "            if slots[j] == -1:\n",
    "                slots[j] = job[0]\n",
    "                profit += job[2]\n",
    "                break\n",
    "\n",
    "    return slots, profit\n",
    "\n",
    "\n",

    "]\n",
    "\n",
