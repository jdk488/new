    "\n",
    "def prim(graph, start):\n",
    "    visited = set()\n",
    "    pq = [(0, start)]\n",
    "    mst = []\n",
    "\n",
    "    while pq:\n",
    "        weight, node = heapq.heappop(pq)\n",
    "\n",
    "        if node not in visited:\n",
    "            visited.add(node)\n",
    "            mst.append((node, weight))\n",
    "\n",
    "            for neighbor, w in graph[node]:\n",
    "                if neighbor not in visited:\n",
    "                    heapq.heappush(pq, (w, neighbor))\n",
    "\n",
    "    return mst\n",
