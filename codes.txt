import re

def read_table(filename):
    """Reads symbol or literal table and returns a dictionary."""
    table = {}
    with open(filename, 'r') as f:
        for line in f:
            parts = line.strip().split("\t")
            if len(parts) >= 3:
                table[int(parts[0])] = int(parts[2])
    return table

def convert(num):
    """Converts number to 3-digit string."""
    num = str(num)
    return num.zfill(3)

def process_pass2():
    pattern = re.compile(r'\(\'([A-Z]{1,2})\',\s*(\d+)\)')
    symbol_table = read_table("symtab.txt")
    literal_table = read_table("littab.txt")

    print("\nSymbol Table:", symbol_table)
    print("Literal Table:", literal_table)
    print("\nLC\tOPCODE\tREG\tOPERAND")

    with open("intermediate.txt", "r") as ic, open("output.txt", "w") as out:
        for line in ic:
            parts = line.strip().split("\t")
            find = pattern.search(parts[0])
            if not find:
                continue

            code_type, code_num = find.groups()
            if code_type in ["IS", "DL"]:
                lc = parts[-2]
                machine_line = lc + "\t" + convert(code_num) + "\t"

                if code_type == "IS":
                    if len(parts) > 2:
                        reg = pattern.search(parts[1])
                        oper = pattern.search(parts[2])
                        reg_val = convert(reg.group(2)) if reg else "000"

                        if oper and oper.group(1) == "S":
                            operand_val = convert(symbol_table[int(oper.group(2))])
                        elif oper and oper.group(1) == "L":
                            operand_val = convert(literal_table[int(oper.group(2))])
                        else:
                            operand_val = "000"

                        machine_line += reg_val + "\t" + operand_val
                    else:
                        machine_line += "000\t000"

                elif code_type == "DL":
                    oper = pattern.search(parts[1])
                    machine_line += "000\t" + (convert(oper.group(2)) if oper else "000")

                print(machine_line)
                out.write(machine_line + "\n")

    print("\n✅ Machine code generated in output.txt")

# Run the assembler
process_pass2()

def pass1():
    with open("input2.txt") as f:
        lines = f.readlines()

    mnt = {}
    mdt = []
    intermediate = []

    inside_macro = False
    macro_name = ""

    for line in lines:
        words = line.strip().split()
        if not words:
            continue

        if words[0] == "MACRO":
            inside_macro = True
            continue

        if inside_macro:
            if words[0] == "MEND":
                mdt.append("MEND")
                inside_macro = False
                continue

            if not macro_name:
                macro_name = words[0]
                mnt[macro_name] = len(mdt)
                mdt.append(" ".join(words[1:]))  # store macro body
            else:
                mdt.append(" ".join(words))
        else:
            intermediate.append(line.strip())

    # Write tables and intermediate output
    with open("MNT.txt", "w") as f:
        for name, index in mnt.items():
            f.write(f"{name}\t{index}\n")

    with open("MDT.txt", "w") as f:
        for i, line in enumerate(mdt):
            f.write(f"{i}\t{line}\n")

    with open("Intermediate.txt", "w") as f:
        for line in intermediate:
            f.write(line + "\n")

    print("\n✅ Pass 1 Done — Created MNT, MDT, and Intermediate File.")

pass1()

            name, index = line.strip().split("\t")
            mnt[name] = int(index)

    with open("MDT.txt") as f:
        for line in f:
            parts = line.strip().split("\t")
            if len(parts) > 1:
                mdt.append(parts[1])

    output = []

    with open("Intermediate.txt") as f:
        for line in f:
            words = line.strip().split()
            if not words:
                continue

            if words[0] in mnt:
                # expand macro

                for i in range(start, len(mdt)):
                    if mdt[i] == "MEND":
                        break
                    output.append(mdt[i].replace("&A", words[1]).replace("&B", words[2]))
            else:
                output.append(line.strip())

    with open("Output.txt", "w") as f:
        for line in output:
            f.write(line + "\n")

    print("\n✅ Pass 2 Done — Expanded macros in Output.txt")

def main():
    # Taking the number of processes
    n = int(input("Enter number of processes: "))
    # Matrix for storing Process ID, Burst Time, Waiting Time & Turnaround Time
    A = [[0 for _ in range(4)] for _ in range(100)]
    total, avg_wt, avg_tat = 0, 0, 0
    print("Enter Burst Time:")
    # User input for Burst Time and assigning Process ID
    for i in range(n):
        A[i][1] = int(input(f"P{i + 1}: "))  # Burst Time
        A[i][0] = i + 1  # Process ID
    # Sorting processes according to their Burst Time
    for i in range(n):
        index = i
        for j in range(i + 1, n):
            if A[j][1] < A[index][1]:
                index = j
        
        # Swap Burst Times
        A[i][1], A[index][1] = A[index][1], A[i][1]
        # Swap Process IDs
        A[i][0], A[index][0] = A[index][0], A[i][0]
    # Initialize waiting time for the first process
    A[0][2] = 0  
    # Calculate waiting times
    for i in range(1, n):
        A[i][2] = sum(A[j][1] for j in range(i))  # Waiting time
        total += A[i][2]
    avg_wt = total / n
    total = 0

    # Calculation of Turnaround Time and printing the data
    print("P\t BT\t WT\t TAT")
    for i in range(n):
        A[i][3] = A[i][1] + A[i][2]  # Turnaround time
        total += A[i][3]
        print(f"P{A[i][0]}\t {A[i][1]}\t {A[i][2]}\t {A[i][3]}")
    avg_tat = total / n
    print(f"Average Waiting Time = {avg_wt}")
    print(f"Average Turnaround Time = {avg_tat}")

def main():
    # Taking the number of processes
    n = int(input("Enter the number of processes: "))

    # Use lists sized to n
    p = [i + 1 for i in range(n)]   # Process IDs
    pp = [0] * n                     # Priorities
    bt = [0] * n                     # Burst Times
    w = [0] * n                      # Waiting Times
    t = [0] * n                      # Turnaround Times

    print("Enter the burst time and priority for each process:")
    for i in range(n):
        print(f"Process[{i + 1}]")
        bt[i] = int(input("  Burst Time: "))
        pp[i] = int(input("  Priority: "))

    # Sort processes by priority (higher number first).
    # We create a list of tuples, sort it, then unpack back to lists.
    # Sorting is stable in Python, so equal priorities preserve original order.
    combined = sorted(zip(pp, bt, p), key=lambda x: x[0], reverse=True)
    pp = [x[0] for x in combined]
    bt = [x[1] for x in combined]
    p  = [x[2] for x in combined]

    # Initialize waiting time and turnaround time
    w[0] = 0
    t[0] = bt[0]
    total_wt = 0
    total_tat = t[0]

    # Calculate waiting and turnaround times for remaining processes
    for i in range(1, n):
        w[i] = t[i - 1]          # waiting time = turnaround time of previous process
        total_wt += w[i]
        t[i] = w[i] + bt[i]      # turnaround time
        total_tat += t[i]

    # Print the process details
    print("\nProcess\tBurst\tWait\tTAT\tPriority")
    for i in range(n):
        print(f"{p[i]}\t{bt[i]}\t{w[i]}\t{t[i]}\t{pp[i]}")

    # Calculate and print average waiting time and turnaround time
    avg_wt = total_wt / n
    avg_tat = total_tat / n
    print(f"\nAverage Wait time: {avg_wt:.2f}")
    print(f"Average Turnaround time: {avg_tat:.2f}")

def findWaitingTime(processes, n, bt, wt, quantum): 
    rem_bt = [0] * n  # Remaining burst times

    # Copy the burst time into rem_bt[] 
    for i in range(n): 
        rem_bt[i] = bt[i]
    t = 0  # Current time 

    while True:
        done = True

        for i in range(n):
            if rem_bt[i] > 0:  # There is a pending process
                done = False  # Mark as not done

                if rem_bt[i] > quantum:
                    t += quantum
                    rem_bt[i] -= quantum 
                else:
                    t += rem_bt[i]
                    wt[i] = t - bt[i]  # Waiting time
                    rem_bt[i] = 0  # Process finished
        # If all processes are done 
        if done:
            break

def findTurnAroundTime(processes, n, bt, wt, tat):
    # Calculating turnaround time 
    for i in range(n):
        tat[i] = bt[i] + wt[i]  # TAT = BT + WT

def findavgTime(processes, n, bt, quantum): 
    wt = [0] * n  # Waiting time
    tat = [0] * n  # Turnaround time 

    findWaitingTime(processes, n, bt, wt, quantum) 
    findTurnAroundTime(processes, n, bt, wt, tat) 

    # Display processes along with all details 
    print("Processes Burst Time  Waiting Time Turn-Around Time")
    total_wt = 0
    total_tat = 0
    for i in range(n):
        total_wt += wt[i] 
        total_tat += tat[i] 
        print(f" {i + 1}\t\t {bt[i]}\t\t {wt[i]}\t\t {tat[i]}")

    avg_wt = total_wt / n
    avg_tat = total_tat / n

    print("\nAverage waiting time = %.2f" % avg_wt)
    print("Average turn around time = %.2f" % avg_tat)

    # Return averages so they can be used later if needed
    return avg_wt, avg_tat

def fifo_page_replacement(pages, capacity):
    frames = []
    page_faults = 0

    print("Page\tFrames\t\tStatus")
    print("-" * 35)

    for page in pages:
        if page not in frames:
            page_faults += 1
            if len(frames) < capacity:
                frames.append(page)
            else:
                frames.pop(0)
                frames.append(page)
            status = "Page Fault"
        else:
            status = "Hit"

        print(f"{page}\t{frames}\t{status}")

    print("\nTotal Page Faults:", page_faults)

from collections import OrderedDict

class LRUCache:
    def __init__(self, capacity):
        self.cache = OrderedDict()
        self.capacity = capacity

    def refer(self, page):
        if page in self.cache:
            # Move page to end (most recently used)
            self.cache.move_to_end(page)
        else:
            # If cache is full, remove least recently used page
            if len(self.cache) >= self.capacity:
                self.cache.popitem(last=False)
            # Add the new page
            self.cache[page] = None


def lru_page_replacement(pages, capacity):
    lru_cache = LRUCache(capacity)
    page_faults = 0

    print("Page\tFrames\t\tStatus")
    print("-" * 35)

    for page in pages:
        if page not in lru_cache.cache:
            page_faults += 1
            status = "Page Fault"
        else:
            status = "Hit"

        # Update cache order (LRU)
        lru_cache.refer(page)

        # Show current memory (frames)
        frames = list(lru_cache.cache.keys())
        print(f"{page}\t{frames}\t{status}")

    print("\nTotal Page Faults:", page_faults)
    print("Total Page Hits:", len(pages) - page_faults)


# ---- Main Program ----

def optimal_page_replacement(pages, capacity):
    if capacity <= 0:
        print("Capacity must be at least 1.")
        return

    page_faults = 0
    page_frames = [-1] * capacity

    print("Page\tFrames\t\t\tStatus")
    print("-" * 45)

    for i, page in enumerate(pages):
        if page not in page_frames:
            page_faults += 1
            # If empty frame exists, use it
            if -1 in page_frames:
                index = page_frames.index(-1)
                page_frames[index] = page
            else:
                # Calculate next occurrence (index) for each page in frames
                future_occurrences = {pf: float('inf') for pf in page_frames}
                for j in range(i + 1, len(pages)):
                    if pages[j] in future_occurrences and future_occurrences[pages[j]] == float('inf'):
                        future_occurrences[pages[j]] = j
                # Replace the page with the farthest next use (or never used -> inf)
                page_to_replace = max(future_occurrences, key=future_occurrences.get)
                index = page_frames.index(page_to_replace)
                page_frames[index] = page

            status = "Page Fault"
        else:
            status = "Hit"

        print(f"{page}\t{page_frames}\t{status}")

    total_hits = len(pages) - page_faults
    print("\nTotal Page Faults:", page_faults)
    print("Total Page Hits:", total_hits)


if __name__ == "__main__":
    pages = list(map(int, input("Enter page reference string: ").split()))
    capacity = int(input("Enter number of frames: "))

    optimal_page_replacement(pages, capacity)

def NextFit(blockSize, m, processSize, n):
		
	allocation = [-1] * n
	j = 0
	t = m-1
	# pick each process and find suitable blocks
	# according to its size ad assign to it
	for i in range(n):

		# Do not start from beginning
		while j < m:
			if blockSize[j] >= processSize[i]:
				
				# allocate block j to p[i] process
				allocation[i] = j
				
				# Reduce available memory in this block.
				blockSize[j] -= processSize[i]
				
				# sets a new end point
				t = (j - 1) % m
				break
			if t == j:
				# sets a new end point
				t = (j - 1) % m
				# breaks the loop after going through all memory block
				break
			
			# mod m will help in traversing the
			# blocks from starting block after
			# we reach the end.
			j = (j + 1) % m
			
	print("Process No. Process Size Block no.")
	
	for i in range(n):
		print("\t", i + 1, "\t", processSize[i],end = "\t\t")

#include "DHT.h"

// IR Sensor Obstacle Detection using LED

int irSensor = 2;   // IR sensor output pin
int led = 13;       // LED pin
int obstacleStatus = 0;

void setup() {
  pinMode(irSensor, INPUT);   // Set IR sensor pin as input
  pinMode(led, OUTPUT);       // Set LED pin as output
  Serial.begin(9600);
}

void loop() {
  obstacleStatus = digitalRead(irSensor);

  if (obstacleStatus == LOW) {   // LOW = Obstacle detected (depends on sensor type)
    digitalWrite(led, HIGH);
    Serial.println("⚠️ Obstacle Detected!");
  } else {
    digitalWrite(led, LOW);
    Serial.println("✅ No Obstacle");
  }

  delay(300);  // Small delay for stability
}


18. Understanding the connectivity of Arduino circuit with temperature sensor. Write 
an application to read the environment temperature and humidity.
#include "DHT.h"

#define DHTPIN 2       // Data pin connected to D2
#define DHTTYPE DHT11  // Sensor type: DHT11

DHT dht(DHTPIN, DHTTYPE);

void setup() {
  Serial.begin(9600);
  Serial.println("DHT11 Sensor Reading...");
  dht.begin();
}

void loop() {
  // Read temperature as Celsius
  float temp = dht.readTemperature();
  // Read humidity
  float hum = dht.readHumidity();

  // Check if any reading failed
  if (isnan(temp) || isnan(hum)) {
    Serial.println("Failed to read from DHT sensor!");
    return;
  }

  Serial.print("🌡 Temperature: ");
  Serial.print(temp);
  Serial.println(" °C");

  Serial.print( Humidity: ");
  Serial.print(hum);
  Serial.println(" %");

  delay(2000);  // Update every 2 seconds
}
